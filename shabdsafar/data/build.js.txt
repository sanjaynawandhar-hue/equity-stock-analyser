for (const line of freqRaw) {
  const w = line.split(' ')[0];
  if (!w || seen.has(w)) continue;
  if (!/^[a-z]{3,24}$/.test(w) && w !== 'a' && w !== 'i') continue;
  if (BLOCK.has(w)) continue;
  seen.add(w);
  freqWords.push(w);
  if (freqWords.length >= 220000) break;
}

// ---------- Webster's ----------
const websters = JSON.parse(fs.readFileSync(path.join(SRC, 'websters.json'), 'utf8'));
const webKeys = new Set(Object.keys(websters));
function webDef(w) {
  let d = websters[w];
  if (!d) return null;
  d = d.replace(/\s+/g, ' ').trim();
  if (/^(Alt\.|See |imp\.|p\. p\.|pl\.|of )/i.test(d)) return null;
  d = d.replace(/^\d+\.\s*/, '');           // leading sense numbering "1. "
  d = d.replace(/\[[^\]]*\]/g, ' ');        // usage tags like [Obs.], [R.]
  d = d.replace(/\s+/g, ' ').trim();
  // cut at first sentence end that isn't an abbreviation
  const m = d.match(/^.{20,190}?[.;](?=\s|$)/);
  d = m ? m[0].replace(/[.;]$/, '') : d.slice(0, 180);
  d = d.trim();
  if (d.length < 20) return null;           // drops junk like "Marrow. Chaucer"
  if (/^[A-Z][a-z]+\.? [A-Z][a-z]+$/.test(d)) return null; // bare cross-refs/attributions
  return d;
}

// ---------- WordNet thesaurus ----------
// Collect every sense; pick the definition from the sense with the most
// synonyms (a decent proxy for the core sense), earliest wins ties.
const wn = new Map(); // word -> {pos, desc, descScore, syns:Set}
const wnLines = fs.readFileSync(path.join(SRC, 'thesaurus.jsonl'), 'utf8').split('\n');
for (const line of wnLines) {
  if (!line.trim()) continue;
  let e; try { e = JSON.parse(line); } catch { continue; }
  const w = e.word;
  if (!/^[a-z]{2,24}$/.test(w)) continue;
  let rec = wn.get(w);
  if (!rec) { rec = { pos: e.pos, desc: null, descScore: -1, syns: new Set() }; wn.set(w, rec); }
  const senseSyns = (e.synonyms || []).length;
  if (e.desc && e.desc.length && senseSyns > rec.descScore) {
    const d = e.desc.find(x => !/^"/.test(x));
    if (d && d.length >= 12) { rec.desc = d.replace(/\s+/g, ' ').trim(); rec.descScore = senseSyns; }
  }
  for (const s of (e.synonyms || [])) {
    const clean = s.replace(/_/g, ' ');
    if (/^[a-z][a-z ]{1,24}$/.test(clean) && clean !== w) rec.syns.add(clean);
  }
}
console.log('wordnet words:', wn.size);

// ---------- IPA ----------
const ipa = new Map();
for (const line of fs.readFileSync(path.join(SRC, 'ipa_en.json'), 'utf8').split('\n')) {
  const [w, p] = line.split('\t');
  if (w && p && !ipa.has(w)) ipa.set(w, p.split(',')[0].trim());
}
console.log('ipa words:', ipa.size);

// ---------- FreeDict TEI ----------
const tei = fs.readFileSync(path.join(SRC, 'eng-hin/eng-hin.tei'), 'utf8');
const entries = tei.split('<entry>').slice(1);
const POS_MAP = { N: 'noun', V: 'verb', Adj: 'adjective', Adv: 'adverb', Det: 'determiner', Prep: 'preposition', Pron: 'pronoun', Conj: 'conjunction', Interj: 'interjection' };
const hindi = new Map(); // word -> {pos, hi:[], ex}
for (const block of entries) {
  const orth = /<orth>([^<]+)<\/orth>/.exec(block);
  if (!orth) continue;
  const w = orth[1].toLowerCase().trim();
  if (!/^[a-z]{2,24}$/.test(w) || hindi.has(w) || BLOCK.has(w)) continue;
  const pos = /<pos>([^<]+)<\/pos>/.exec(block);
  const trans = [];
  const transRe = /<cit type="trans">\s*<quote>([^<]+)<\/quote>/g;
  let m;
  while ((m = transRe.exec(block)) && trans.length < 3) {
    // strip FreeDict domain tags {...} and expand alternates [...] into " / "
    const parts = m[1].replace(/\{[^}]*\}/g, '').replace(/~/g, ' ').replace(/\[/g, '/').replace(/\]/g, '')
      .split('/').map(x => x.replace(/\s+/g, ' ').trim()).filter(Boolean);
    for (const p of parts) {
      if (/[ऀ-ॿ]/.test(p) && !trans.includes(p) && trans.length < 3) trans.push(p);
    }
  }
  if (!trans.length) continue;
  const ex = /<cit type="example">\s*<quote>([^<]+)<\/quote>/.exec(block);
  hindi.set(w, { pos: POS_MAP[pos ? pos[1] : ''] || '', hi: trans, ex: ex ? ex[1].replace(/\s+/g, ' ').trim() : '' });
}
console.log('hindi entries:', hindi.size);

// ---------- seed cards ----------
// card: [word, pos, ipa, enDef, hiMeaning, example, [syns], ant, band]
function defFor(w) {
  const rec = wn.get(w);
  if (rec && rec.desc && rec.desc.length >= 12 && rec.desc.length <= 200) return rec.desc;
  return webDef(w);
}
function synsFor(w) {
  const rec = wn.get(w);
  if (!rec) return [];
  return [...rec.syns].filter(s => s !== w).slice(0, 3);
}
const cards = [];
const carded = new Set();
function tryCard(w) {
  if (carded.has(w)) return;
  const h = hindi.get(w);
  if (!h) return;
  const def = defFor(w);
  if (!def) return;
  const r = rankOf.has(w) ? rankOf.get(w) : 65000;
  cards.push([w, h.pos || (wn.get(w) ? wn.get(w).pos : ''), ipa.get(w) || '', def, h.hi.slice(0, 3).join(' / '), h.ex || '', synsFor(w), ANT[w] || '', bandOf(r)]);
  carded.add(w);
}
// themed words first, then by frequency rank
